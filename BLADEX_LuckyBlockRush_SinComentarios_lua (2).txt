local function findKnit()
    local ok, idx = pcall(function()
        return RS.Packages._Index
    end)
    if not ok then return nil end
    for _, folder in ipairs(idx:GetChildren()) do
        if folder.Name:lower():match("knit") then
            local knitFolder = folder:FindFirstChild("knit")
            if knitFolder then
                local services = knitFolder:FindFirstChild("Services")
                if services then return services end
            end
        end
    end
    return nil
end

local knitServices = findKnit()

local function findRF(serviceName, rfName)
    if not knitServices then return nil end
    local svc = knitServices:FindFirstChild(serviceName)
    if not svc then return nil end
    local rf = svc:FindFirstChild("RF")
    if not rf then return nil end
    return rf:FindFirstChild(rfName)
end

local function findRE(serviceName, reName)
    if not knitServices then return nil end
    local svc = knitServices:FindFirstChild(serviceName)
    if not svc then return nil end
    local re = svc:FindFirstChild("RE")
    if not re then return nil end
    return re:FindFirstChild(reName)
end

local function safeInvoke(remote, ...)
    if not remote then return false, "remoto nulo" end
    local ok, res = pcall(remote.InvokeServer, remote, ...)
    return ok, res
end

local function safeFireRE(remote, ...)
    if not remote then return false end
    local ok = pcall(remote.FireServer, remote, ...)
    return ok
end

local activeThreads = {}

local function startLoop(name, fn)
    if activeThreads[name] then
        pcall(task.cancel, activeThreads[name])
        activeThreads[name] = nil
    end
    activeThreads[name] = task.spawn(fn)
end

local function stopLoop(name)
    if activeThreads[name] then
        pcall(task.cancel, activeThreads[name])
        activeThreads[name] = nil
    end
end

local function equipWeapon()
    pcall(function()
        local char = LocalPlayer.Character
        local backpack = LocalPlayer.Backpack
        if not char or not backpack then return end

        for _, item in ipairs(char:GetChildren()) do
            if item:IsA("Tool") and item.Name ~= "Weapon" then
                item.Parent = backpack
            end
        end

        local weapon = backpack:FindFirstChild("Weapon")
        if weapon then
            weapon.Parent = char
        end
    end)
end

local hooked = {}
local lastFire = 0
local COOLDOWN = 0.1

local function fireButton(btn)
    local now = tick()
    if now - lastFire < COOLDOWN then return end
    lastFire = now
    if getconnections then
        pcall(function()
            for _, conn in ipairs(getconnections(btn.Activated)) do
                pcall(conn.Function)
            end
            for _, conn in ipairs(getconnections(btn.MouseButton1Click)) do
                pcall(conn.Function)
            end
        end)
    end
    if firesignal then
        pcall(function()
            firesignal(btn.Activated)
            firesignal(btn.MouseButton1Click)
        end)
    end
end

local function isX2Button(obj)
    if not obj:IsA("ImageButton") then return false end
    if obj.Name ~= "Button" then return false end
    local parent = obj.Parent
    if not parent then return false end
    return string.find(string.lower(parent.Name), "x2") ~= nil
end

local function hookObj(obj)
    if hooked[obj] then return end
    if not isX2Button(obj) then return end
    hooked[obj] = true
    if obj.Visible then
        fireButton(obj)
    end
    obj:GetPropertyChangedSignal("Visible"):Connect(function()
        if not _G.AutoTraining then return end
        if obj.Visible then
            fireButton(obj)
        end
    end)
    local parent = obj.Parent
    if parent then
        parent:GetPropertyChangedSignal("Visible"):Connect(function()
            if not _G.AutoTraining then return end
            if parent.Visible then
                local btn = parent:FindFirstChild("Button")
                if btn and btn:IsA("ImageButton") then
                    fireButton(btn)
                end
            end
        end)
    end
end

    end)
end

local function loopAutoCollectCash()
    local cachedPlotId = nil

    local function getMyPlotId()
        if cachedPlotId then return cachedPlotId end
        if rfGetMyPlotId then
            local ok, res = pcall(rfGetMyPlotId.InvokeServer, rfGetMyPlotId)
            if ok and res then cachedPlotId = tostring(res) return cachedPlotId end
        end
        local plots = workspace:FindFirstChild("Plots")
        if plots then
            local uid = LocalPlayer.UserId
            for _, plotModel in ipairs(plots:GetChildren()) do
                local owner = plotModel:GetAttribute("OwnerId")
                    or plotModel:GetAttribute("Owner")
                    or plotModel:GetAttribute("PlayerId")
                if owner == uid then cachedPlotId = plotModel.Name return cachedPlotId end
                for _, sub in ipairs(plotModel:GetChildren()) do
                    local subOwner = sub:GetAttribute("OwnerId")
                        or sub:GetAttribute("Owner")
                        or sub:GetAttribute("PlayerId")
                    if subOwner == uid then cachedPlotId = plotModel.Name return cachedPlotId end
                end
            end
        end
        return nil
    end

    local function touchPad(pad)
        if not pad then return end
        pcall(function()
            local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            local cd = pad:FindFirstChildOfClass("ClickDetector")
            if cd then fireclickdetector(cd) return end
            local pp = pad:FindFirstChildOfClass("ProximityPrompt")
            if pp then fireproximityprompt(pp) return end
            if hrp then
                firetouchinterest(pad, hrp, 0)
                task.wait(0.04)
                firetouchinterest(pad, hrp, 1)
            end
        end)
    end

    local function getPads(plotId)
        local pads = {}
        local plots = workspace:FindFirstChild("Plots")
        if not plots then return pads end
        local plotModel = plots:FindFirstChild(plotId)
        if not plotModel then return pads end
        local subPlot = plotModel:FindFirstChild(plotId) or plotModel:GetChildren()[1]
        if not subPlot then return pads end
        local containers = subPlot:FindFirstChild("Containers")
        if not containers then return pads end
        for slot = 1, 30 do
            local slotStr = tostring(slot)
            local slotCont = containers:FindFirstChild(slotStr)
            if slotCont then
                local subCont = slotCont:FindFirstChild(slotStr) or slotCont:GetChildren()[1]
                if subCont then
                    local collection = subCont:FindFirstChild("Collection")
                    if collection then
                        local pad = collection:FindFirstChild("CollectionPad")
                        if pad then table.insert(pads, pad) end
                    end
                end
            end
        end
        return pads
    end

    while _G.AutoCollectCash do
        local plotId = getMyPlotId()
        if not plotId then cachedPlotId = nil task.wait(5) continue end
        local pads = getPads(plotId)
        for _, pad in ipairs(pads) do
            if not _G.AutoCollectCash then break end
            task.spawn(function() touchPad(pad) end)
        end
        task.wait(5)
    end
end

    if godCharWatcher then pcall(function() godCharWatcher:Disconnect() end) godCharWatcher = nil end

    local char = LocalPlayer.Character

local function applyGodMode(char)
    if not char then return end
    local hum = char:WaitForChild("Humanoid", 8)
    if not hum then return end
    pcall(function() hum.MaxHealth = math.huge hum.Health = math.huge end)
    if godHumConn then pcall(function() godHumConn:Disconnect() end) end
    godHumConn = hum.HealthChanged:Connect(function(hp)
        if _GodMode and hp < math.huge then
            pcall(function() hum.Health = math.huge end)
        end
    end)
end

local function loopGodMode()
    godCharWatcher = LocalPlayer.CharacterAdded:Connect(function(newChar)
        if _GodMode then task.spawn(function() applyGodMode(newChar) end) end
    end)
    applyGodMode(LocalPlayer.Character)
    local lastChar = LocalPlayer.Character
    while _GodMode do
        local char = LocalPlayer.Character
        if char and char ~= lastChar then
            lastChar = char
            task.spawn(function() applyGodMode(char) end)
        end
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function()
                    if hum.MaxHealth ~= math.huge then hum.MaxHealth = math.huge end
                    if hum.Health < 9e17 then hum.Health = math.huge end
                end)
            end
        end
        task.wait(0.1)
    end
    stopGodMode()
end

local suffixes = {
    ["K"]  = 1e3,  ["M"]  = 1e6,  ["B"]  = 1e9,
    ["T"]  = 1e12, ["Qa"] = 1e15, ["Qi"] = 1e18,
    ["Sx"] = 1e21, ["Sp"] = 1e24,
}

local function parseValue(str)
    if not str then return 0 end
    str = tostring(str):gsub("%$",""):gsub("%s","")
    for suffix, mult in pairs(suffixes) do
        local num = str:match("^([%d%.]+)" .. suffix .. "$")
        if num then return tonumber(num) * mult end
    end
    return tonumber(str) or 0
end

local rarityWeight = {
    ["Normal"] = 1, ["Candy"] = 2, ["Gold"] = 3,
    ["Diamond"] = 4, ["Void"] = 5,
}

local function getRarityWeight(str)
    if not str then return 1 end
    for k, v in pairs(rarityWeight) do
        if str:lower() == k:lower() then return v end
    end
    return 1
end

local function hasMutation(skillTagText)
    if not skillTagText then return false end
    local t = skillTagText:lower():gsub("%s","")
    return t ~= "sell$" and t ~= "sell" and t ~= ""
end

local SHELL_LIMIT  = 190
local SHELL_TARGET = 189

local rfSellBrainrot = findRF("InventoryService", "SellBrainrot")

local function getScrollingFrame()
    local pg      = LocalPlayer:FindFirstChild("PlayerGui")
    local windows = pg and pg:FindFirstChild("Windows")
    local sb      = windows and windows:FindFirstChild("SellBrainrots")
    local sc      = sb and sb:FindFirstChild("ShopContainer")
    return sc and sc:FindFirstChild("ScrollingFrame")
end

local function countInventory()
    local sf = getScrollingFrame()
    if not sf then return 0 end
    local count = 0
    for _, card in ipairs(sf:GetChildren()) do
        if card.Name ~= "UIGridLayout" and card.Name ~= "UIListLayout" then
            count = count + 1
        end
    end
    return count
end

local function readInventory()
    local sf = getScrollingFrame()
    if not sf then return nil, 0 end
    local result = {}
    local count  = 0
    for _, card in ipairs(sf:GetChildren()) do
        if card.Name ~= "UIGridLayout" and card.Name ~= "UIListLayout" then
            count = count + 1
            local main = card:FindFirstChild("Main")
            if main then
                local rarityTxt = (main:FindFirstChild("Rarity")    and main.Rarity.Text)    or "Normal"
                local cashTxt   = (main:FindFirstChild("Cash")      and main.Cash.Text)      or "0$"
                local nameTxt   = (main:FindFirstChild("TextLabel") and main.TextLabel.Text) or card.Name
                local skillTxt  = (main:FindFirstChild("SkillTag")  and main.SkillTag.Text)  or "Sell $"
                local levelTxt  = (main:FindFirstChild("Level")     and main.Level.Text)     or "Lvl 1"
                table.insert(result, {
                    uid      = card.Name,
                    name     = nameTxt,
                    rarity   = rarityTxt,
                    weight   = getRarityWeight(rarityTxt),
                    cash     = parseValue(cashTxt),
                    cashText = cashTxt,
                    mutation = hasMutation(skillTxt),
                    skillTag = skillTxt,
                    level    = levelTxt,
                })
            end
        end
    end
    return result, count
end

local function selectToSell(inventory, needed)
    local noMut = {}
    for _, b in ipairs(inventory) do
        if not b.mutation then table.insert(noMut, b) end
    end
    table.sort(noMut, function(a, b)
        if a.weight ~= b.weight then return a.weight < b.weight end
        return a.cash < b.cash
    end)
    local toSell = {}
    for i = 1, math.min(needed, #noMut) do
        table.insert(toSell, noMut[i])
    end
    return toSell
end

local function doShell()
    local inv, total = readInventory()
    if not inv then return end
    if total < SHELL_LIMIT then return end
    local needed = total - SHELL_TARGET
    if needed <= 0 then return end
    local toSell = selectToSell(inv, needed)
    if #toSell == 0 then return end
    for _, b in ipairs(toSell) do
        if not rfSellBrainrot then break end
        pcall(rfSellBrainrot.InvokeServer, rfSellBrainrot, b.uid)
        task.wait(0.15)
    end
end

local function loopAutoShellInteligente()
    local lastCount = countInventory()
    while _G.AutoShellInteligente do
        local current = countInventory()
        if current ~= lastCount then
            lastCount = current
            if current >= SHELL_LIMIT then
                pcall(doShell)
                lastCount = countInventory()
            end
        end
        task.wait(0.5)
    end
end
